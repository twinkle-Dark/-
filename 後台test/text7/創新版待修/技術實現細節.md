  bindChartTypeButtons: function() {
      // 圖表類型切換
      chartTypeButtons.forEach(button => {
          button.addEventListener('click', (e) => {
              e.preventDefault();

              const chartType = button.getAttribute('data-chart-type');
              const chartContainer = button.closest('.card').querySelector('canvas');

              if (chartContainer && chartContainer.chart) {
                  // 儲存當前數據

                  // 銷毀當前圖表
                  chartContainer.chart.destroy();

                  // 創建新類型的圖表

                      type: chartType,
                      data: currentData,
                      options: this.getChartOptions(chartType)
                  });

                  chartContainer.chart = newChart;
              }
          });
      });
  }

### 1. 趨勢分析引擎

- **線性回歸分析**

  ```javascript
  generateTrendAnalysis: function(trendData) {
      // ...

      // 分析趨勢方向（上升、下降或穩定）
      // 使用簡單的線性回歸斜率來確定趨勢
      let trend = '穩定';
      let slope = 0;
      let validDataPoints = 0;

      const filteredData = [];
      for (let i = 0; i < validMonthData.length; i++) {
          if (validMonthData[i] > 0) {
              filteredData.push({ x: i, y: validMonthData[i] });
              validDataPoints++;
          }
      }

      if (validDataPoints >= 3) {
          // 計算均值
          const avgX = filteredData.reduce((sum, point) => sum + point.x, 0) / filteredData.length;
          const avgY = filteredData.reduce((sum, point) => sum + point.y, 0) / filteredData.length;

          // 計算斜率
          let numerator = 0;
          let denominator = 0;

          for (const point of filteredData) {
              const xDiff = point.x - avgX;
              const yDiff = point.y - avgY;
              numerator += xDiff * yDiff;
              denominator += xDiff * xDiff;
          }

          if (denominator !== 0) {
              slope = numerator / denominator;

              // 判定趨勢方向
              if (slope > 0.1) {
                  trend = '上升';
              } else if (slope < -0.1) {
                  trend = '下降';
              }
          }
      }

      analysis.push({
          title: '整體趨勢',
          value: trend
      });

      // ...
  }
  ```
